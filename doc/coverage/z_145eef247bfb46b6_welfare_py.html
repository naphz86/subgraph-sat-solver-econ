12def _check_unit(name: str, value: Rational) -> Fraction: 

13 v = Fraction(value) 

14 if not 0 <= v <= 1: 

15 raise ValueError(f"{name} muss in [0, 1] liegen") 

16 return v

19def computation_gap(optimum: Rational, achieved: Rational) -> Fraction: 

20 """Rechenlücke ``g = (opt − A)/opt ∈ [0,1]``.""" 

21 opt, got = Fraction(optimum), Fraction(achieved) 

22 if opt <= 0: 

23 raise ValueError("opt muss positiv sein") 

24 if not 0 <= got <= opt: 

25 raise ValueError("A muss in [0, opt] liegen") 

26 return (opt - got) / opt

29def total_output(optima: Sequence[Rational], gaps: Sequence[Rational]) -> Fraction: 

30 """``Y = Σ_i opt_i (1 − g_i)``.""" 

31 if len(optima) != len(gaps): 

32 raise ValueError("optima und gaps müssen gleich lang sein") 

33 return sum((Fraction(o) * (1 - _check_unit("g", g)) for o, g in zip(optima, gaps)), Fraction(0))

36def welfare_gain(alpha: Rational, mean_gap: Rational) -> Fraction: 

37 """Relativer Zugewinn ``(O − Y)/Y = αγ̄ / (1 − αγ̄)`` (Satz ``satz:gain``). 

38 

39 ``alpha``: Wertanteil der NP-artigen Entscheidungen an ``O``; ``mean_gap``: wertgewichtete 

40 mittlere Lücke dieser Entscheidungen. 

41 """ 

42 a, g = _check_unit("alpha", alpha), _check_unit("mean_gap", mean_gap) 

43 product = a * g 

44 if product == 1: 

45 raise ValueError("α·γ̄ = 1: Output Y wäre 0, der relative Zugewinn undefiniert") 

46 return product / (1 - product)

49def net_gain(alpha: Rational, mean_gap: Rational, replacement_cost_share: Rational) -> Fraction: 

50 """Zugewinn abzüglich Ersatzkosten der Sicherheitsinfrastruktur (Kapitel 8, nichtmonoton). 

51 

52 ``replacement_cost_share``: Ersatzkosten als Anteil an ``Y``. Rein modellhaft: die Arbeit 

53 liefert keinen Zahlenwert, die Parameter sind empirisch zu bestimmen. 

54 """ 

55 return welfare_gain(alpha, mean_gap) - _check_unit("replacement_cost_share", replacement_cost_share)

63def communication_requirement(num_goods: int) -> int: 

64 """Zwei Agenten, ``L`` unteilbare Güter, allgemeine Bewertungen (Satz ``satz:comm``): 

65 ein Preis je nichtleerem Bündel, also ``2^L − 1`` Werte müssen offenbart werden.""" 

66 if num_goods < 0: 

67 raise ValueError("Anzahl der Güter muss nichtnegativ sein") 

68 return (1 << num_goods) - 1
